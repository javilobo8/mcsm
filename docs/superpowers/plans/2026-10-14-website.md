# Project website Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An eye-catching SPA in `website/` that presents MC Server Manager and carries its full documentation, deployed to GitHub Pages by `.github/workflows/pages.yml`.

**Architecture:** Vite + React SPA with path routes under `/mcsm/`. Docs are Markdown compiled at build time by a local Vite plugin into a virtual module (HTML + TOC); a post-build step writes one `index.html` per route, `404.html`, `sitemap.xml` and `robots.txt`. Two tracks: App (code, landing, docs shell) and Content (Markdown pages), merged by the controller.

**Tech Stack:** React 19, TypeScript strict, Vite 8, Tailwind 4, shadcn 3.8.5 (Radix), lucide-react, motion, TanStack Router, cmdk, unified/remark/rehype, shiki, github-slugger, zod 4, vitest 4.

**Spec:** `docs/superpowers/specs/2026-10-14-website-design.md` (read it first; it is the authority).

// build/docs/frontmatter.ts
export const SECTIONS = ["Getting started", "Guides", "Operations", "Project"] as const;
export type Section = (typeof SECTIONS)[number];
export interface DocMeta { title: string; description: string; section: Section; order: number }
/** Splits YAML frontmatter from the body and validates it with zod. Errors name `file`. */
export function parseDoc(source: string, file: string): { meta: DocMeta; body: string };

// build/docs/compile.ts
export interface TocEntry { depth: 2 | 3; id: string; text: string }
export interface CompileOptions {
  base: string;                       // ends with "/", e.g. "/mcsm/"
  repoRoot: string;                   // absolute path of the repository root
  readFile?: (absPath: string) => string; // default fs.readFileSync(p, "utf8")
  file?: string;                      // for error messages
}
export interface CompiledDoc {
  html: string;
  toc: TocEntry[];        // h2 and h3 only, in order
  headingIds: string[];   // every heading id (h2–h6)
  links: string[];        // internal links as written: "/docs/<slug>" or "/docs/<slug>#<anchor>"
  images: string[];       // image src as written: "/screenshots/<name>.webp"
}
export function compileDoc(body: string, opts: CompileOptions): Promise<CompiledDoc>;

// build/docs/load.ts
export interface LoadedDoc extends DocMeta, CompiledDoc { slug: string }
/** Reads every *.md in `dir` (missing dir → []), parses, compiles, sorts with sortPages. */
export function loadDocs(dir: string, opts: Omit<CompileOptions, "file">): Promise<LoadedDoc[]>;

// src/lib/docs-nav.ts (pure, used by the app and the build)
export interface NavPage { slug: string; title: string; section: string; order: number }
export function sortPages<T extends NavPage>(pages: T[]): T[];          // by order; duplicate order → throws naming both slugs
export function groupBySection<T extends NavPage>(pages: T[]): { section: string; pages: T[] }[]; // sections in order of their first page
export function neighbours<T extends NavPage>(pages: T[], slug: string): { prev?: T; next?: T }; // over the sorted list

// src/lib/search.ts (pure)
export interface SearchEntry { slug: string; anchor?: string; title: string; page: string; section: string; text: string }
export function buildSearchIndex(pages: { slug: string; title: string; description: string; section: string; toc: { id: string; text: string }[] }[]): SearchEntry[];
export function search(index: SearchEntry[], query: string, limit?: number): SearchEntry[]; // default limit 8

// virtual:docs (src/virtual-docs.d.ts)
declare module "virtual:docs" {
  export interface DocPage { slug: string; title: string; description: string; section: string; order: number; html: string; toc: { depth: 2 | 3; id: string; text: string }[] }
  export const pages: DocPage[]; // sorted
}
```

Compile behaviour: remark-parse + remark-gfm + remark-frontmatter-free body → include plugin → remark-rehype (allowDangerousHtml false) → rehype-slug (GitHub slugs) → autolink headings (an anchor `#` link inside each h2/h3) → alerts → link/image rewrite → shiki (`github-dark-default` or similar dark theme; languages bash, shell, yaml, ini, dotenv, json, ts, go, text; unknown language → plain text) → stringify. Code blocks are wrapped as `<div class="code-block" data-lang="<lang>">…shiki pre…</div>`. Alerts become `<div class="callout callout-<kind>"><p class="callout-title">Note|Tip|Important|Warning|Caution</p>…</div>`. Internal links `/docs/...` → `${base}docs/<slug>/` + `#anchor`; images `/screenshots/...` → `${base}screenshots/...` with `loading="lazy"`; external `http(s)` links get `target="_blank" rel="noopener noreferrer"`.

// test/docs-nav.test.ts
import { describe, it, expect } from "vitest";
import { sortPages, groupBySection, neighbours } from "../src/lib/docs-nav";

const p = (slug: string, section: string, order: number) => ({ slug, title: slug, section, order });
const pages = [p("backups", "Guides", 50), p("intro", "Getting started", 10), p("install", "Getting started", 20), p("security", "Operations", 90)];

describe("docs-nav", () => {
  it("sorts by order", () => {
    expect(sortPages(pages).map((x) => x.slug)).toEqual(["intro", "install", "backups", "security"]);
  });
  it("throws on a duplicate order naming both pages", () => {
    expect(() => sortPages([p("a", "Guides", 1), p("b", "Guides", 1)])).toThrow(/a.*b|b.*a/);
  });
  it("groups sections in the order of their first page", () => {
    expect(groupBySection(sortPages(pages)).map((g) => [g.section, g.pages.map((x) => x.slug)])).toEqual([
      ["Getting started", ["intro", "install"]],
      ["Guides", ["backups"]],
      ["Operations", ["security"]],
    ]);
  });
  it("gives prev/next across sections and none at the ends", () => {
    const s = sortPages(pages);
    expect(neighbours(s, "install")).toEqual({ prev: s[0], next: s[2] });
    expect(neighbours(s, "intro").prev).toBeUndefined();
    expect(neighbours(s, "security").next).toBeUndefined();
    expect(neighbours(s, "nope")).toEqual({});
  });
});

import { describe, it, expect } from "vitest";
import { buildSearchIndex, search } from "../src/lib/search";

const index = buildSearchIndex([
  { slug: "backups", title: "Backups", description: "Hot and scheduled backups, restore.", section: "Guides",
    toc: [{ id: "restoring-and-pre-restore-data", text: "Restoring and pre-restore data" }, { id: "space", text: "Space, files and simultaneous jobs" }] },
  { slug: "router", title: "Router (mc-router)", description: "Join servers by hostname.", section: "Guides",
    toc: [{ id: "autoscale", text: "Autoscale" }] },
]);

describe("search", () => {
  it("indexes pages and their headings", () => {
    expect(index.filter((e) => e.slug === "backups")).toHaveLength(3);
    expect(index.find((e) => e.anchor === "autoscale")).toMatchObject({ page: "Router (mc-router)", title: "Autoscale" });
  });
  it("is case-insensitive and ranks title matches first", () => {
    const r = search(index, "BACKUPS");
    expect(r[0]).toMatchObject({ slug: "backups", anchor: undefined });
  });
  it("requires every word to match", () => {
    expect(search(index, "restore data").map((e) => e.anchor)).toContain("restoring-and-pre-restore-data");
    expect(search(index, "restore autoscale")).toEqual([]);
  });
  it("returns [] for blank or unmatched queries and honours the limit", () => {
    expect(search(index, "   ")).toEqual([]);
    expect(search(index, "zzzz")).toEqual([]);
    expect(search(index, "e", 2)).toHaveLength(2);
  });
});
```

Word matching is on prefixes or substrings of `title + page + text` (text = description for page entries, the heading for heading entries); rank: title starts with query > title contains > other; ties keep index order (page entry before its headings).
